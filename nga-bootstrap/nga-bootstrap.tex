\section{Introduction: Quantum Many-body Bootstrap}

\begin{frame}{Example: SHO}
  Consider a simple harmonic oscillator:
  \[H=\frac12p^2+\frac12x^2.\]
  \begin{itemize}
  \item Classically: $\langle H\rangle\geq 0$, $\langle H\rangle=0$ when $\langle x\rangle=\langle p\rangle = 0$.
  \item QM: $\Delta x\cdot \Delta p\geq\hbar$, $\langle H\rangle\geq \frac12\hbar$.
  \end{itemize}
  \begin{columns}
    \column{.5\textwidth}
    \begin{block}{Formal proof.}<2->
      \[\langle (p+ix)(p-ix)\rangle\geq 0.\]
      \[\langle p^2\rangle+\langle x^2\rangle\geq -i\langle [x,p]\rangle =\hbar.\]
      \[\therefore \langle H\rangle\geq \frac\hbar2.\]
    \end{block}
    \column{.4\textwidth}
    % \begin{block}{Essential factors}
    Essential factors:
      \begin{itemize}
      \item<3-> $\langle O^\dagger O\rangle \geq 0.$
      \item<4-> Commutation relation.
      \item<5-> Minimize $\langle H\rangle$.
      \end{itemize}
    %\end{block}
  \end{columns}
\end{frame}

\begin{frame}{Bootstrap algorithm}

  \begin{columns}
    \column{.4\textwidth}
    \begin{block}{Minimization problem}
      \begin{align*}
        E=\min_{f} \quad &\langle H\rangle,\\
        \mathrm{s.t.} \quad &\langle O^\dagger O\rangle \geq 0.
      \end{align*}
      where $f: O\mapsto\langle O\rangle$.
    \end{block}

    % \begin{center}
    \centering\includegraphics{relax-method}
    %\end{center}
    \column{.6\textwidth}
    Bootstrap is a relaxation method:
    \begin{itemize}
    \item Givens \textbf{lower bound} of the ground-state energy.
    \item \textbf{Compliments} instead of \textbf{competes} with variational method such as DMRG.
    \item Together with DMRG, can give \textbf{rigorous bounds} to the ground-state energy and other observables.
    \end{itemize}
  \end{columns}
  
\end{frame}

\begin{frame}{Truncation and Approximation}
  \begin{block}{Truncation}
    \begin{itemize}
    \item \textbf{A set of basis:} $O_i, i=1,\ldots,N$; $\mathcal V=\sspan\{O_i\}$.
    \item Basis are strings of Majorana-fermion operators:
      $O=\gamma_{i_1}\cdots\gamma_{i_n}$.
    \item $\mathcal M:=\mathcal V^\dagger\mathcal V$. $\symbfit x\in \mathcal M^\ast$: a functional $O\mapsto \langle O\rangle$.
    \item $M(\symbfit x)$: a $N\times N$ matrix $M_{ij}:=\langle O_i^\dagger O_j\rangle.$
    \end{itemize}
  \end{block}

  \begin{block}{Symmetry constraints}
    \begin{itemize}
    \item Continuous symmetry: $\left< [\mathcal Q, O]\right>=0$;
      in particular, $\left<[H,O]\right>=0$.
    \item Discrete symmetry: $\left< O^a\right>=0$ unless $a$ is a trivial irrep of $G$.
    \end{itemize}
  \end{block}
  \[E=\min_{\symbfit x\in\mathcal M^\ast} \symbfit c^T\symbfit x,\quad \mathrm{s.t.} \quad M\succeq0; A\symbfit x=\symbfit b.\]\end{frame}

  \begin{columns}

\section{Method: Nullspace-Guided Adaptive (NGA) Bootstrap}

\begin{frame}{Motivation: null vector and annihilator of GS}
  \begin{columns}
    \column{.5\textwidth}
    \begin{itemize}
    \item Convex optimization: $\min \langle H\rangle$ must lie on the boundary of the feasible set.
    \item On the boundary, $M$ has null vector(s): $P\in \mathcal V$ s. t. $\langle P^\dagger P\rangle_{\symbfit x^\ast}=0$.
    \item $\langle \mathrm{GS}|P^\dagger P|\mathrm{GS}\rangle=0$ $\Rightarrow$ $P|\mathrm{GS}\rangle =0$.
    \item $P$ annihilates $|\mathrm{GS}\rangle$.
    \item Null vectors = GS annihilators, and they are essential to the bootstrap.
    \end{itemize}
    \column{.5\textwidth}
    \begin{block}{Example: SHO}
      \begin{itemize}
      \item Bootstrap: $\langle(p+ix)(p-ix)\rangle\geq0 \Rightarrow \langle H\rangle\geq\frac{\hbar}2$.
      \item Annihilator: $p-ix=a$, $a|0\rangle=0$.
      \end{itemize}
    \end{block}
    \begin{center}
      \includegraphics{psd-cone}
    \end{center}
  \end{columns}
  
\end{frame}

\begin{frame}{Null-space guided bootstrap}
  \begin{columns}
    \column{.4\textwidth}
        \begin{tikzpicture}[
        node distance=0.45cm and 0.35cm,
        box/.style={
            draw,
            rounded corners,
            align=center,
            font=\scriptsize,
            inner sep=3pt,
            minimum width=2.0cm,
            minimum height=0.55cm
        },
        arrow/.style={-{Latex[length=1.6mm]}, semithick}
    ]
    \node[box] (init) {Initialize\\[1pt]$\mathcal{B}^{(0)}$};
    \node[box, below=of init] (solve) {Solve SDP};
    \node[box, below=of solve] (null) {Extract nullspace\\of $M(\symbfit{x}^\star)$};
    \node[box, below left=0.4cm and -0.5cm of null] (prune) {Prune by $\ell_a$};
    \node[box, below right=0.4cm and -0.5cm of null] (grow) {Grow by $s_\mu$};
    \node[box, below=1.35cm of null] (update) {Update\\[1pt]$\mathcal{B}^{(n+1)}$};

    \draw[arrow] (init) -- (solve);
    \draw[arrow] (solve) -- (null);
    \draw[arrow] (null) -- (prune);
    \draw[arrow] (null) -- (grow);
    \draw[arrow] (prune) -- (update);
    \draw[arrow] (grow) -- (update);
    \draw[arrow] (update.east) -- ++(1.85cm,0) |- (solve.east);
    \end{tikzpicture}
    \column{.6\textwidth}
    \begin{itemize}
    \item Operators in null-space are important.
    \item Eliminate $O_i$ that has little overlap with null vectors.
    \item Add $[H, P_\alpha]$ to $\mathcal V$.
      \begin{itemize}
      \item If $P_\alpha|\mathrm{GS}\rangle=0$, then $[H,P_\alpha]|\mathrm{GS}\rangle=0$ as well.
      \item $[H, P_\alpha]$ is needed to describe the dynamics of $P_\alpha$.
      \end{itemize}
    \item Iterate and grow the size of $\mathcal V$.
    \end{itemize}
  \end{columns}
\end{frame}

\section{Conclusion and future directions}

\begin{frame}{Conclusion and future directions}
  The current work:
  \begin{itemize}
  \item[\checkmark] Nullspace-Guided Adaptive (NGA) Bootstrap can sysmetically improve the lower bound.
  \item[\checkmark] 2-3 orders of magnitude improvement comparing to previous works.
  \item[\checkmark] Favorable scaling w/ the size of $\mathcal V$ and the associated SDP.
  \end{itemize}
  Future directions:
  \begin{itemize}
  \item[?] The bottleneck is now the SDP solving efficiency and memory consumption: better solver tailored to the Bootstrap problem?\\

  \item[?] 2d problems, finite-T, dynamics, ...
  \item[?] Use MPO or TPO to represent the operators in the null space?
  \end{itemize}
\end{frame}
